cmake_minimum_required(VERSION 3.22.1)
project(paper_canon C)
file(READ "${CMAKE_CURRENT_LIST_DIR}/gutenprint/sources.json" SOURCE_JSON)
string(JSON SOURCE_COUNT LENGTH "${SOURCE_JSON}")
math(EXPR LAST_SOURCE "${SOURCE_COUNT} - 1")
set(CANON_SOURCES paper_canon.c)
foreach(INDEX RANGE ${LAST_SOURCE})
    string(JSON SOURCE GET "${SOURCE_JSON}" ${INDEX})
    list(APPEND CANON_SOURCES "gutenprint/${SOURCE}")
endforeach()
add_library(paper_canon SHARED ${CANON_SOURCES})
target_include_directories(paper_canon PRIVATE gutenprint gutenprint/include gutenprint/src)
target_compile_definitions(paper_canon PRIVATE HAVE_CONFIG_H=1 _GNU_SOURCE)
target_compile_options(paper_canon PRIVATE -O2 -fvisibility=hidden)
target_link_options(paper_canon PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
target_link_libraries(paper_canon m)
